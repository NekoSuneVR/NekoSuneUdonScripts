using UdonSharp;
using UnityEngine;
using UnityEngine.UI;
using VRC.SDK3.Data;
using VRC.SDK3.Image;
using VRC.SDK3.StringLoading;
using VRC.SDKBase;
using VRC.Udon.Common.Interfaces;

[UdonBehaviourSyncMode(BehaviourSyncMode.None)]
public class NekoImageGalleryRuntime : UdonSharpBehaviour
{
    [Header("Images")]
    public Texture[] localTextures;
    public VRCUrl[] imageUrls;
    public int sourceMode; // 0 local textures, 1 predeclared VRCUrl array

    [Header("Optional JSON metadata")]
    public VRCUrl metadataJsonUrl;
    public string rootKey = "items";
    public string titleKey = "title";
    public string subtitleKey = "subtitle";
    public string imageIndexKey = "imageIndex";
    public string imageUrlKey = "imageUrl";

    [Header("UI")]
    public RawImage frontImage;
    public RawImage backImage;
    public RawImage shaderImage;
    public CanvasGroup frontGroup;
    public CanvasGroup backGroup;
    public Text titleText;
    public Text subtitleText;
    public Text pageText;
    public Text statusText;

    [Header("Transitions")]
    public int transitionMode; // 0 fade, 1 slide left, 2 slide right, 3 slide up, 4 zoom, 5 spin zoom, 6 wipe, 7 dissolve, 8 radial
    public float transitionDuration = 0.65f;
    public Material transitionMaterial;
    public bool autoPlay;
    public float autoPlaySeconds = 7f;

    private VRCImageDownloader _downloader;
    private IVRCImageDownload[] _downloads;
    private Texture[] _remoteTextures;
    private string[] _titles;
    private string[] _subtitles;
    private int[] _mappedIndices;
    private int _metadataCount;
    private int _index;
    private int _targetIndex;
    private int _pendingRemoteIndex = -1;
    private float _transitionTime;
    private float _autoTime;
    private bool _transitioning;
    private Vector2 _frontBase;
    private Vector2 _backBase;

    void Start()
    {
        _downloader = new VRCImageDownloader();
        int remoteCount = imageUrls == null ? 0 : imageUrls.Length;
        _downloads = new IVRCImageDownload[remoteCount];
        _remoteTextures = new Texture[remoteCount];
        _frontBase = frontImage == null ? Vector2.zero : frontImage.rectTransform.anchoredPosition;
        _backBase = backImage == null ? Vector2.zero : backImage.rectTransform.anchoredPosition;
        if (shaderImage != null) shaderImage.gameObject.SetActive(false);
        if (backGroup != null) backGroup.alpha = 0f;
        if (!VRCUrl.IsNullOrEmpty(metadataJsonUrl)) VRCStringDownloader.LoadUrl(metadataJsonUrl, this);
        ShowInitial();
    }

    void Update()
    {
        if (_transitioning) UpdateTransition();
        if (autoPlay && !_transitioning)
        {
            _autoTime += Time.deltaTime;
            if (_autoTime >= Mathf.Max(1f, autoPlaySeconds)) { _autoTime = 0f; Next(); }
        }
    }

    public void Next()
    {
        int count = ItemCount(); if (count <= 1 || _transitioning) return;
        ShowIndex((_index + 1) % count);
    }

    public void Previous()
    {
        int count = ItemCount(); if (count <= 1 || _transitioning) return;
        int n = _index - 1; if (n < 0) n = count - 1; ShowIndex(n);
    }

    public void ToggleAutoPlay()
    {
        autoPlay = !autoPlay; _autoTime = 0f;
        SetStatus(autoPlay ? "Auto-play enabled" : "Auto-play paused");
    }

    public void ReloadMetadata()
    {
        if (VRCUrl.IsNullOrEmpty(metadataJsonUrl)) { SetStatus("No metadata JSON URL configured"); return; }
        SetStatus("Loading metadata...");
        VRCStringDownloader.LoadUrl(metadataJsonUrl, this);
    }

    public override void OnStringLoadSuccess(IVRCStringDownload result)
    {
        DataToken root;
        if (!VRCJson.TryDeserializeFromJson(result.Result, out root)) { SetStatus("JSON parse failed: " + root.ToString()); return; }
        DataList list = null;
        if (root.TokenType == TokenType.DataList) list = root.DataList;
        else if (root.TokenType == TokenType.DataDictionary)
        {
            DataDictionary d = root.DataDictionary; DataToken token;
            if (!string.IsNullOrEmpty(rootKey) && d.TryGetValue(rootKey, TokenType.DataList, out token)) list = token.DataList;
            else if (d.TryGetValue("items", TokenType.DataList, out token)) list = token.DataList;
            else if (d.TryGetValue("images", TokenType.DataList, out token)) list = token.DataList;
            else if (d.TryGetValue("gallery", TokenType.DataList, out token)) list = token.DataList;
            else if (d.TryGetValue("data", TokenType.DataList, out token)) list = token.DataList;
        }
        if (list == null) { SetStatus("JSON does not contain an array at the root/items/images/gallery/data"); return; }

        int count = Mathf.Min(list.Count, 256);
        _titles = new string[count]; _subtitles = new string[count]; _mappedIndices = new int[count];
        for (int i = 0; i < count; i++)
        {
            _mappedIndices[i] = i;
            DataToken item;
            if (!list.TryGetValue(i, TokenType.DataDictionary, out item)) continue;
            DataDictionary row = item.DataDictionary;
            _titles[i] = GetString(row, titleKey, "title", "name", "label");
            _subtitles[i] = GetString(row, subtitleKey, "subtitle", "description", "caption");
            DataToken n;
            if (!string.IsNullOrEmpty(imageIndexKey) && row.TryGetValue(imageIndexKey, out n) && n.IsNumber)
                _mappedIndices[i] = Mathf.RoundToInt((float)n.Double);
            else
            {
                string url = GetString(row, imageUrlKey, "imageUrl", "image", "url", "thumbnailImageUrl");
                int mapped = FindPredeclaredUrl(url); if (mapped >= 0) _mappedIndices[i] = mapped;
            }
        }
        _metadataCount = count; _index = 0;
        SetStatus("Loaded " + count + " gallery entries");
        ShowInitial();
    }

    public override void OnStringLoadError(IVRCStringDownload result)
    {
        SetStatus("Metadata download failed: " + result.ErrorCode + " - " + result.Error);
    }

    public override void OnImageLoadSuccess(IVRCImageDownload result)
    {
        int idx = FindPredeclaredUrl(result.Url.Get());
        if (idx < 0 || idx >= _remoteTextures.Length) return;
        _remoteTextures[idx] = result.Result;
        if (_pendingRemoteIndex == idx)
        {
            _pendingRemoteIndex = -1;
            BeginTransition(result.Result, _targetIndex);
        }
    }

    public override void OnImageLoadError(IVRCImageDownload result)
    {
        _pendingRemoteIndex = -1;
        SetStatus("Image failed: " + result.Error + " - " + result.ErrorMessage);
    }

    void ShowInitial()
    {
        int count = ItemCount();
        if (count <= 0) { SetStatus("Assign localTextures or predeclared imageUrls"); UpdateLabels(0); return; }
        _index = Mathf.Clamp(_index, 0, count - 1);
        int sourceIndex = SourceIndexFor(_index);
        Texture tex = GetTexture(sourceIndex);
        if (tex != null)
        {
            if (frontImage != null) frontImage.texture = tex;
            UpdateLabels(_index);
        }
        else if (sourceMode == 1) RequestRemote(sourceIndex, _index, false);
    }

    void ShowIndex(int next)
    {
        int count = ItemCount(); if (count <= 0) return;
        next = (next % count + count) % count;
        int sourceIndex = SourceIndexFor(next);
        Texture tex = GetTexture(sourceIndex);
        _targetIndex = next;
        if (tex != null) BeginTransition(tex, next);
        else if (sourceMode == 1) RequestRemote(sourceIndex, next, true);
    }

    void RequestRemote(int sourceIndex, int displayIndex, bool transition)
    {
        if (imageUrls == null || sourceIndex < 0 || sourceIndex >= imageUrls.Length || VRCUrl.IsNullOrEmpty(imageUrls[sourceIndex])) { SetStatus("Image URL mapping is missing"); return; }
        if (_remoteTextures[sourceIndex] != null)
        {
            if (transition) BeginTransition(_remoteTextures[sourceIndex], displayIndex);
            else { if (frontImage != null) frontImage.texture = _remoteTextures[sourceIndex]; _index = displayIndex; UpdateLabels(_index); }
            return;
        }
        if (_downloads[sourceIndex] != null) _downloads[sourceIndex].Dispose();
        _targetIndex = displayIndex; _pendingRemoteIndex = sourceIndex;
        SetStatus("Loading image " + (displayIndex + 1) + "...");
        _downloads[sourceIndex] = _downloader.DownloadImage(imageUrls[sourceIndex], null, (IUdonEventReceiver)this, new TextureInfo());
    }

    void BeginTransition(Texture nextTexture, int nextIndex)
    {
        if (frontImage == null || nextTexture == null) return;
        if (backImage == null || frontGroup == null || backGroup == null)
        {
            frontImage.texture = nextTexture; _index = nextIndex; UpdateLabels(_index); return;
        }
        backImage.texture = nextTexture; backImage.rectTransform.anchoredPosition = _backBase; backImage.rectTransform.localScale = Vector3.one; backImage.rectTransform.localRotation = Quaternion.identity;
        frontImage.rectTransform.anchoredPosition = _frontBase; frontImage.rectTransform.localScale = Vector3.one; frontImage.rectTransform.localRotation = Quaternion.identity;
        frontGroup.alpha = 1f; backGroup.alpha = 0f; _transitionTime = 0f; _targetIndex = nextIndex; _transitioning = true;
        if (transitionMode >= 6 && shaderImage != null && transitionMaterial != null)
        {
            transitionMaterial.SetTexture("_FromTex", frontImage.texture); transitionMaterial.SetTexture("_ToTex", nextTexture); transitionMaterial.SetFloat("_Progress", 0f); transitionMaterial.SetFloat("_Mode", transitionMode - 6); shaderImage.material = transitionMaterial; shaderImage.gameObject.SetActive(true);
        }
    }

    void UpdateTransition()
    {
        float duration = Mathf.Max(0.05f, transitionDuration);
        _transitionTime += Time.deltaTime;
        float t = Mathf.Clamp01(_transitionTime / duration);
        float e = t * t * (3f - 2f * t);
        RectTransform fr = frontImage.rectTransform; RectTransform br = backImage.rectTransform;

        if (transitionMode == 0) { frontGroup.alpha = 1f - e; backGroup.alpha = e; }
        else if (transitionMode == 1) { float w = fr.rect.width; fr.anchoredPosition = _frontBase + Vector2.left * w * e; br.anchoredPosition = _backBase + Vector2.right * w * (1f - e); backGroup.alpha = 1f; }
        else if (transitionMode == 2) { float w = fr.rect.width; fr.anchoredPosition = _frontBase + Vector2.right * w * e; br.anchoredPosition = _backBase + Vector2.left * w * (1f - e); backGroup.alpha = 1f; }
        else if (transitionMode == 3) { float h = fr.rect.height; fr.anchoredPosition = _frontBase + Vector2.up * h * e; br.anchoredPosition = _backBase + Vector2.down * h * (1f - e); backGroup.alpha = 1f; }
        else if (transitionMode == 4) { fr.localScale = Vector3.one * Mathf.Lerp(1f, 1.18f, e); br.localScale = Vector3.one * Mathf.Lerp(0.82f, 1f, e); frontGroup.alpha = 1f - e; backGroup.alpha = e; }
        else if (transitionMode == 5) { fr.localScale = Vector3.one * Mathf.Lerp(1f, 0.65f, e); fr.localRotation = Quaternion.Euler(0f, 0f, 30f * e); br.localScale = Vector3.one * Mathf.Lerp(1.35f, 1f, e); br.localRotation = Quaternion.Euler(0f, 0f, -30f * (1f - e)); frontGroup.alpha = 1f - e; backGroup.alpha = e; }
        else if (shaderImage != null && transitionMaterial != null) { transitionMaterial.SetFloat("_Progress", e); shaderImage.color = Color.white; frontGroup.alpha = 0f; backGroup.alpha = 0f; }
        else { frontGroup.alpha = 1f - e; backGroup.alpha = e; }

        if (t >= 1f) FinishTransition();
    }

    void FinishTransition()
    {
        frontImage.texture = backImage.texture; frontImage.rectTransform.anchoredPosition = _frontBase; frontImage.rectTransform.localScale = Vector3.one; frontImage.rectTransform.localRotation = Quaternion.identity;
        frontGroup.alpha = 1f; backGroup.alpha = 0f; backImage.rectTransform.anchoredPosition = _backBase; backImage.rectTransform.localScale = Vector3.one; backImage.rectTransform.localRotation = Quaternion.identity;
        if (shaderImage != null) shaderImage.gameObject.SetActive(false);
        _index = _targetIndex; _transitioning = false; _autoTime = 0f; UpdateLabels(_index); SetStatus("");
    }

    int ItemCount()
    {
        if (_metadataCount > 0) return _metadataCount;
        if (sourceMode == 0) return localTextures == null ? 0 : localTextures.Length;
        return imageUrls == null ? 0 : imageUrls.Length;
    }

    int SourceIndexFor(int displayIndex)
    {
        if (_metadataCount > 0 && _mappedIndices != null && displayIndex >= 0 && displayIndex < _mappedIndices.Length) return _mappedIndices[displayIndex];
        return displayIndex;
    }

    Texture GetTexture(int sourceIndex)
    {
        if (sourceMode == 0) return localTextures != null && sourceIndex >= 0 && sourceIndex < localTextures.Length ? localTextures[sourceIndex] : null;
        return _remoteTextures != null && sourceIndex >= 0 && sourceIndex < _remoteTextures.Length ? _remoteTextures[sourceIndex] : null;
    }

    void UpdateLabels(int displayIndex)
    {
        int count = ItemCount();
        if (pageText != null) pageText.text = count <= 0 ? "0 / 0" : (displayIndex + 1) + " / " + count;
        if (titleText != null) titleText.text = _titles != null && displayIndex < _titles.Length && !string.IsNullOrEmpty(_titles[displayIndex]) ? _titles[displayIndex] : "Gallery Image " + (displayIndex + 1);
        if (subtitleText != null) subtitleText.text = _subtitles != null && displayIndex < _subtitles.Length ? _subtitles[displayIndex] : "";
    }

    string GetString(DataDictionary d, string preferred, string a, string b, string c)
    {
        DataToken token;
        if (!string.IsNullOrEmpty(preferred) && d.TryGetValue(preferred, out token) && token.TokenType == TokenType.String) return token.String;
        if (d.TryGetValue(a, out token) && token.TokenType == TokenType.String) return token.String;
        if (d.TryGetValue(b, out token) && token.TokenType == TokenType.String) return token.String;
        if (d.TryGetValue(c, out token) && token.TokenType == TokenType.String) return token.String;
        return "";
    }

    int FindPredeclaredUrl(string value)
    {
        if (string.IsNullOrEmpty(value) || imageUrls == null) return -1;
        for (int i = 0; i < imageUrls.Length; i++) if (!VRCUrl.IsNullOrEmpty(imageUrls[i]) && imageUrls[i].Get() == value) return i;
        return -1;
    }

    void SetStatus(string value) { if (statusText != null) statusText.text = value; }

    void OnDestroy()
    {
        if (_downloads != null) for (int i = 0; i < _downloads.Length; i++) if (_downloads[i] != null) _downloads[i].Dispose();
        if (_downloader != null) _downloader.Dispose();
    }
}
